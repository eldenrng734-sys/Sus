import json

import httpx
from pydantic import ValidationError

from app.modules.ai.service import ConversationAnalysisOutput

OPENAI_CHAT_COMPLETIONS_URL = "https://api.openai.com/v1/chat/completions"


class AIProviderError(RuntimeError):
    """The configured AI provider could not return a valid analysis."""


def analyze_with_openai(
    prompt: dict[str, str],
    *,
    api_key: str,
    model: str,
    timeout_seconds: float,
) -> ConversationAnalysisOutput:
    try:
        with httpx.Client(timeout=timeout_seconds) as client:
            response = client.post(
                OPENAI_CHAT_COMPLETIONS_URL,
                headers={"Authorization": f"Bearer {api_key}"},
                json={
                    "model": model,
                    "messages": [
                        {"role": "system", "content": prompt["system"]},
                        {"role": "user", "content": prompt["user"]},
                    ],
                    "response_format": {"type": "json_object"},
                    "temperature": 0.1,
                },
            )
            response.raise_for_status()
            payload = response.json()
    except httpx.HTTPError as exc:
        raise AIProviderError("The AI provider request failed") from exc
    except ValueError as exc:
        raise AIProviderError("The AI provider returned invalid JSON") from exc

    try:
        content = payload["choices"][0]["message"]["content"]
        if not isinstance(content, str):
            raise TypeError("response content is not text")
        return ConversationAnalysisOutput.model_validate(json.loads(content))
    except (IndexError, KeyError, TypeError, ValueError, ValidationError) as exc:
        raise AIProviderError("The AI provider returned an invalid conversation analysis") from exc

@pytest.fixture(scope="session")
def tokens(
    admin_client: TestClient,
    seed: Seed,
) -> dict[tuple[str, str], dict[str, str]]:
    """Log in every seeded user and complete MFA enrollment for admins."""
    result: dict[tuple[str, str], dict[str, str]] = {}

    for letter in ("a", "b"):
        for key, role, _active in USER_PLAN:
            email = seed.get(letter).users[key][1]

            # Step 1: Log in.
            response = admin_client.post(
                "/api/v1/auth/login",
                json={"email": email, "password": PASSWORD},
            )
            assert response.status_code == 200, f"Login failed for {letter}/{key}: {response.text}"

            login_data = response.json()["data"]

            # Step 2: Admins must complete MFA before receiving a token.
            if role == "admin":
                assert login_data.get("mfa_setup_required") is True, (
                    f"Expected MFA enrollment for {letter}/{key}, received: {login_data}"
                )

                challenge = login_data["challenge"]

                # Start MFA enrollment and retrieve the TOTP secret.
                setup_response = admin_client.post(
                    "/api/v1/auth/mfa/setup",
                    json={"challenge": challenge},
                )
                assert setup_response.status_code == 200, (
                    f"MFA setup failed for {letter}/{key}: {setup_response.text}"
                )

                secret = setup_response.json()["data"]["secret"]

                # Generate a valid time-based one-time password.
                code = pyotp.TOTP(secret).now()

                # Confirm enrollment and obtain the authenticated session.
                confirm_response = admin_client.post(
                    "/api/v1/auth/mfa/confirm",
                    json={
                        "challenge": challenge,
                        "code": code,
                    },
                )
                assert confirm_response.status_code == 200, (
                    f"MFA confirmation failed for {letter}/{key}: {confirm_response.text}"
                )

                token = confirm_response.json()["data"]["access_token"]

            else:
                # Managers and reps use the normal login response.
                token = login_data["access_token"]

            result[(letter, key)] = {"Authorization": f"Bearer {token}"}

    return result
